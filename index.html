import os import subprocess import whisper def tao_phu_de_hardsub(video_path, output_path="output_hardsub.mp4"): print("1. Đang trích xuất và nhận diện giọng nói từ video bằng Whisper...") # Load model whisper (có thể đổi thành 'base', 'small', 'medium' tùy độ mạnh yếu của máy) model = whisper.load_model("base") # Chạy AI nhận diện sub result = model.transcribe(video_path) # Tạo file phụ đề .srt từ kết quả nhận diện srt_path = "sub_tam_thoi.srt" with open(srt_path, "w", encoding="utf-8") as f: for i, segment in enumerate(result["segments"], start=1): start_time = dinh_dang_thoi_gian(segment["start"]) end_time = dinh_dang_thoi_gian(segment["end"]) text = segment["text"].strip() f.write(f"{i}\n{start_time} --> {end_time}\n{text}\n\n") print(f"2. Đã tạo xong file phụ đề tạm: {srt_path}") print("3. Đang tiến hành ép phụ đề cứng (hardsub) vào video qua FFmpeg...") # Dùng ffmpeg ép sub cứng vào video # Lưu ý: đường dẫn file srt cần được xử lý dấu gạch chéo nếu chạy trên Windows cmd = [ "ffmpeg", "-y", "-i", video_path, "-vf", f"subtitles={srt_path}:force_style='FontSize=24,PrimaryColour=&H00FFFF&'", output_path ] subprocess.run(cmd, check=True) print(f"4. Xong xuôi! Video hardsub đã lưu tại: {output_path}") def dinh_dang_thoi_gian(seconds): hours = int(seconds // 3600) minutes = int((seconds % 3600) // 60) secs = int(seconds % 60) milliseconds = int((seconds - int(seconds)) * 1000) return f"{hours:02d}:{minutes:02d}:{secs:02d},{milliseconds:03d}" if __name__ == "__main__": # Thay tên file video của bạn vào đây video_input = "video_cua_ban.mp4" if os.path.exists(video_input): tao_phu_de_hardsub(video_input) else: print(f"Không tìm thấy file video {video_input} kìa bạn ơi!")
